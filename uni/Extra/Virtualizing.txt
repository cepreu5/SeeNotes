Здравейте! Това е отличен въпрос, който засяга един от ключовите проблеми с производителността на приложението. Виртуализацията на списъка е най-добрата техника за справяне със забавянето при рендиране на голям брой елементи.

Какво е "виртуализация" и защо е нужна?
Проблемът: Когато имате стотици или хиляди бележки, приложението в момента създава HTML елемент за всяка една от тях. Дори бележките да са извън екрана, браузърът трябва да ги обработи, изчисли стиловете им и да ги пази в паметта. Това е изключително бавно и ресурсоемко, което води до голямото първоначално забавяне.

Решението (Виртуализация): Вместо да рендираме всички 1000+ елемента, ние "лъжем" браузъра:

Създаваме един голям, празен контейнер, чиято височина е равна на височината, която биха заели всички бележки, ако бяха рендирани. Това създава работещ скролбар.
Създаваме малък брой HTML елементи (например за 20-30 бележки), които ще преизползваме.
Докато потребителят скролира, ние изчисляваме кои бележки трябва да са видими в момента.
Динамично пълним нашия малък набор от елементи с данните на видимите бележки и ги позиционираме абсолютно вътре в големия контейнер.
Така браузърът работи само с малък брой елементи по всяко време, което прави приложението изключително бързо, независимо от общия брой бележки.

Как може да се имплементира в main.js?
Имплементацията на виртуализация от нулата е сложна задача. Тя изисква значително преструктуриране на начина, по който се показват бележките. Ето основните стъпки и примерен код как би изглеждала логиката.

Необходими промени:

Нови глобални променливи: Ще ни трябват променливи за състоянието на виртуализацията.
Промяна в renderUI: Тази функция вече няма да създава всички бележки, а ще подготви контейнера за виртуализация и ще извика първоначалното рендиране.
Нова функция renderVisibleNotes: Това ще е сърцето на системата. Тя ще се извиква при скролиране и ще актуализира видимите елементи.
Промяна в applyFilters: Филтрирането вече няма да скрива/показва елементи (display: none), а ще работи върху масив с данни, който след това ще се подава на renderVisibleNotes.
Добавяне на scroll event listener: Ще следим скролирането на прозореца.

function createNoteElementPool() {
    noteElementPool.forEach(el => el.remove()); // Изчистваме стария басейн

function handleVirtualScroll() {
    // Използваме requestAnimationFrame за по-добра производителност
    window.requestAnimationFrame(() => {
        // Рендираме само ако скролът се е променил достатъчно
        if (Math.abs(window.scrollY - lastRenderedScrollTop) > NOTE_HEIGHT / 2) {
            renderVisibleNotes();
            lastRenderedScrollTop = window.scrollY;
        }
    });
}

async function renderVisibleNotes() {
    const scrollTop = window.scrollY;
    const viewportHeight = window.innerHeight;
    const notesPerRow = Math.max(1, Math.floor(notesContainer.clientWidth / NOTE_WIDTH));

    // Изчисляваме кои редове са видими
    const firstVisibleRow = Math.max(0, Math.floor(scrollTop / NOTE_HEIGHT) - OVERSCAN_ROWS);
    const lastVisibleRow = Math.min(
        Math.ceil(virtualNotesData.length / notesPerRow),
        Math.floor((scrollTop + viewportHeight) / NOTE_HEIGHT) + OVERSCAN_ROWS
    );

    // Изчисляваме индексите на бележките, които трябва да се покажат
    const startIndex = firstVisibleRow * notesPerRow;
    const endIndex = Math.min(virtualNotesData.length, lastVisibleRow * notesPerRow);

    const visibleData = virtualNotesData.slice(startIndex, endIndex);
    const visibleIndices = new Set();

    // Обновяваме елементите от басейна
    for (let i = 0; i < visibleData.length; i++) {
        const dataIndex = startIndex + i;
        const noteData = virtualNotesData[dataIndex];
        const poolIndex = dataIndex % noteElementPool.length; // Взимаме елемент от басейна
        const noteEl = noteElementPool[poolIndex];
        visibleIndices.add(poolIndex);

        // Позиционираме елемента
        const row = Math.floor(dataIndex / notesPerRow);
        const col = dataIndex % notesPerRow;
        noteEl.style.top = `${row * NOTE_HEIGHT}px`;
        noteEl.style.left = `${col * NOTE_WIDTH}px`;
        noteEl.style.display = 'flex';

        // Попълваме съдържанието (адаптирана логика от createNoteElement)
        // Вместо да създаваме нов елемент, обновяваме съществуващия
        await updateNoteElement(noteEl, noteData.rawData);
    }

    // Скриваме неизползваните елементи от басейна
    for (let i = 0; i < noteElementPool.length; i++) {
        if (!visibleIndices.has(i)) {
            noteElementPool[i].style.display = 'none';
        }
    }
}

function applyFilters() {
    const searchTerm = searchBox.value.toLowerCase();
    
    // 1. Филтрираме `allNotesData`
    virtualNotesData = allNotesData.filter(noteData => {
        const data = noteData.content; // Вече имаме парснатите данни
        const isVisibleByBoard = (currentBoardFilter === 'all') ||
                                 (currentBoardFilter === 'reminder' && data.timer && data.timer !== 0) ||
                                 (data.boardid === currentBoardFilter);

        const isVisibleBySearch = (() => {
            if (!searchTerm) return true;
            if (searchMode === 'title') {
                // Трябва да извлечем заглавието от notetxt
                const noteTitle = (data.notetxt || "").split('\n')[0].toLowerCase();
                return noteTitle.includes(searchTerm);
            } else { // searchMode === 'content'
                return (data.notetxt || "").toLowerCase().includes(searchTerm);
            }
        })();

        return isVisibleByBoard && isVisibleBySearch;
    });

    // 2. Сортираме филтрираните данни, ако е нужно
    if (localStorage.getItem('enableNoteSorting') === 'true') {
        virtualNotesData.sort((a, b) => {
            const numordA = (a.content.numord !== undefined && a.content.numord !== null) ? a.content.numord : Infinity;
            const numordB = (b.content.numord !== undefined && b.content.numord !== null) ? b.content.numord : Infinity;
            return numordA - numordB;
        });
    }

    // 3. Актуализираме UI за виртуализацията
    // Преизчисляваме височината на контейнера и рендираме видимите бележки

    const noteCounter = document.getElementById('note-counter');
    if (noteCounter) {
        noteCounter.textContent = virtualNotesData.length; // Показваме броя на филтрираните
    }
}

За да подновите токена, трябва да използвате "refresh token". Ето как работи процесът:

Получаване на Refresh Token: При първоначалното влизане на потребителя, вашето приложение трябва да поиска от Google offline достъп. В отговор, освен access_token, Google ще върне и refresh_token. Този refresh_token е дълготраен и се използва, за да се получават нови access_token-и.

Съхраняване на Refresh Token: Трябва да съхраните refresh_token на сигурно място. Тъй като sessionStorage се изтрива при затваряне на браузъра, може да искате да използвате localStorage, ако искате потребителят да остане логнат за по-дълго.

Подновяване на Access Token: Когато access_token е напът да изтече (както вашият checkAuth засича), вместо да препращате към login.html, трябва да изпратите POST заявка към https://oauth2.googleapis.com/token със следните параметри:

client_id: Вашият клиентски ID от Google.
client_secret: Вашият клиентски "secret" ключ. ВНИМАНИЕ: Съхраняването на client_secret в клиентски JavaScript код е много рисковано и не се препоръчва!
refresh_token: Запазеният refresh token.
grant_type: refresh_token.
Актуализиране на токена: В отговор на заявката ще получите нов access_token (с нов срок на валидност). Трябва да го запазите в sessionStorage (или localStorage), заменяйки стария.

Примерна функция за подновяване:

async function refreshToken() {
    const storedTokenString = sessionStorage.getItem('google_auth_token'); // Or localStorage
    if (!storedTokenString) throw new Error('No token found');
    
    const tokenData = JSON.parse(storedTokenString);
    const refreshToken = tokenData.refresh_token;

    if (!refreshToken) {
        // If there is no refresh token, we can't refresh.
        // This might happen if 'offline' access was not requested.
        throw new Error('No refresh token available.');
    }

    const response = await fetch('https://oauth2.googleapis.com/token', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/x-www-form-urlencoded',
        },
        body: new URLSearchParams({
            'client_id': 'YOUR_CLIENT_ID', // Replace with your actual client ID
            'client_secret': 'YOUR_CLIENT_SECRET', // Replace with your actual client secret - **SECURITY RISK**
            'refresh_token': refreshToken,
            'grant_type': 'refresh_token'
        })
    });

    if (!response.ok) {
        throw new Error('Failed to refresh token');
    }

    const newAuthData = await response.json();
    
    // Google might not return a new refresh_token, so preserve the old one.
    newAuthData.refresh_token = refreshToken; 
    newAuthData.issued_at = Date.now(); // Add issued_at timestamp

    sessionStorage.setItem('google_auth_token', JSON.stringify(newAuthData));
    return newAuthData;
}

async function checkAuth() {
    const storedTokenString = sessionStorage.getItem('google_auth_token');
    if (!storedTokenString) {
        window.location.href = 'login.html';
        return;
    }
    
    let authToken = JSON.parse(storedTokenString);
    const isExpired = (Date.now() - authToken.issued_at) / 1000 > (authToken.expires_in - 60);

    if (isExpired) {
        try {
            console.log('Access token expired, attempting to refresh...');
            authToken = await refreshToken();
            console.log('Token refreshed successfully.');
        } catch (error) {
            console.error('Failed to refresh token:', error);
            sessionStorage.removeItem('google_auth_token');
            showToast(_('errorSessionExpired'));
            window.location.href = 'login.html';
        }
    }
    // Continue with the application logic using the (potentially new) authToken
}
Важни съображения:
