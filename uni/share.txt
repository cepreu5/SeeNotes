self.addEventListener("fetch", event => {
  const url = new URL(event.request.url);

  if (url.pathname === "/share" && event.request.method === "POST") {
    event.respondWith(handleShare(event.request));
  }
});

async function handleShare(request) {
  const formData = await request.formData();

  const title = formData.get("title") || "";
  const text = formData.get("text") || "";
  const url = formData.get("url") || "";

  // Пакетираме данните в URL параметри
  const params = new URLSearchParams({
    shared: "1",
    title,
    text,
    url
  });

  // Пренасочваме към MultiNotes UI
  return Response.redirect("/?" + params.toString(), 303);
}

service-worker.js — обработка на файловете

self.addEventListener("fetch", event => {
  const url = new URL(event.request.url);

  if (url.pathname === "/share" && event.request.method === "POST") {
    event.respondWith(handleShare(event.request));
  }
});

async function handleShare(request) {
  const formData = await request.formData();

  const title = formData.get("title") || "";
  const text = formData.get("text") || "";
  const url = formData.get("url") || "";

  // Файловете могат да бъдат един или много
  const files = formData.getAll("file");

  // Съхраняваме ги временно в IndexedDB или Cache Storage
  // За минимален пример — просто ги сериализираме в sessionStorage
  const payload = {
    title,
    text,
    url,
    files: await Promise.all(files.map(fileToObject))
  };

  sessionStorage.setItem("sharedPayload", JSON.stringify(payload));

  return Response.redirect("/?shared=1", 303);
}

async function fileToObject(file) {
  return {
    name: file.name,
    type: file.type,
    size: file.size,
    data: await file.arrayBuffer()
  };
}

В твоя MultiNotes UI — приемане на файловете

if (location.search.includes("shared=1")) {
  const raw = sessionStorage.getItem("sharedPayload");

  if (raw) {
    const payload = JSON.parse(raw);
    sessionStorage.removeItem("sharedPayload");

    handleSharedData(payload);
  }
}

Обработка на споделените файлове

function handleSharedData({ title, text, url, files }) {
  let content = "";

  if (title) content += title + "\n\n";
  if (text) content += text + "\n\n";
  if (url) content += url + "\n\n";

  // Създаваме нова бележка
  const noteId = createNewNote({
    content,
    created: Date.now(),
    source: "share"
  });

  // Добавяме файловете към бележката
  for (const file of files) {
    saveFileToNote(noteId, file);
  }

  openNoteEditor(noteId);
}
