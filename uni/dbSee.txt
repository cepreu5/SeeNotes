indexedDB.open("NotesDB").onsuccess = function(event) {
  const db = event.target.result;
  const tx = db.transaction("notes", "readonly");
  const store = tx.objectStore("notes");
  const req = store.getAll();
  req.onsuccess = () => console.log(req.result);
};

indexedDB.open("NotesDB").onsuccess = function(event) {
  const db = event.target.result;
  const tx = db.transaction("notes", "readonly");
  const store = tx.objectStore("notes");
  const request = store.openCursor(); // използваме курсор, за да вземем само първия запис

  request.onsuccess = function() {
    const cursor = request.result;
    if (cursor) {
      console.log("Първи запис:", cursor.value);
    } else {
      console.log("Няма записи в store-а.");
    }
  };

  request.onerror = function() {
    console.error("Грешка при четене от IndexedDB:", request.error);
  };
};
